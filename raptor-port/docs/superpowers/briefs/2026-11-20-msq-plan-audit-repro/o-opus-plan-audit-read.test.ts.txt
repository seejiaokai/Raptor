const mem: Record<string, string> = {}
const fake = {
  getItem: (k: string) => (k in mem ? mem[k]! : null),
  setItem: (k: string, v: string) => { if (v === 'null') delete mem[k]; else mem[k] = v },
  keys: () => Object.keys(mem),
}
const signInAs = (name: string, pass: string) => { const r = signIn(name, pass); resetSession(sessionFor(r)); notify(); return r }
let ISNAP = ''

beforeEach(() => {
  vi.useFakeTimers({ toFake: ['Date'] })
  vi.setSystemTime(new Date(2026, 6, 15, 9, 0, 0))                    // Wed 15 Jul 26
  if (!ISNAP) ISNAP = JSON.stringify(INPUTS)
  INPUTS.length = 0; JSON.parse(ISNAP).forEach((r: any) => INPUTS.push(r))
  Object.keys(mem).forEach(k => delete mem[k])
  storeBackend.impl = fake
  setUnitSeeds(true); initStore(); accountsLoad(); resetSession(null); _resetTimeline(); installGlobalUndo()
})
afterEach(() => { setSession(null); resetSession(null); _resetTimeline(); vi.useRealTimers() })
afterAll(() => { storeBackend.impl = null })

describe('O1 — a home move is never one a Plan holder may make of himself or of a master admin (D2205, D166 pattern)', () => {
  it('(a) Saber gives Hunter Plan of 149; Hunter moves HIMSELF into 149: refused', () => {
    signInAs('ad', 'a')
    expect(setGrant('acad2', 'u149', 'admin', 'prowler').ok, 'setup: Plan given by the master admin').toBe(true)
    signInAs('ad2', 'a2')
    const r = setHome('prowler', 'u149')
    unitsLoad()
    expect(r.ok, `his own move: ${JSON.stringify(r)}; his squadron tomorrow ${homeOn('prowler', '2026-07-16')}`).toBe(false)
  })
  it('(b) Saber gives Hunter Plan of 149; Hunter moves SABER, the master admin, into 149: refused (D2205 — "M3 holds against a planner too")', () => {
    signInAs('ad', 'a')
    expect(setGrant('acad2', 'u149', 'admin', 'prowler').ok, 'setup').toBe(true)
    signInAs('ad2', 'a2')
    const r = setHome('stiff', 'u149')
    unitsLoad()
    expect(r.ok, `the master admin moved by a planner: ${JSON.stringify(r)}; Saber tomorrow ${homeOn('stiff', '2026-07-16')}`).toBe(false)
  })
  it('(control) Saber moves Hunter into 149: allowed', () => {
    signInAs('ad', 'a')
    expect(setHome('prowler', 'u149').ok).toBe(true)
  })
})

describe('O2 — Approvals\' "new since you last looked" is kept for each squadron he opens (D1231 with D2197, D2201)', () => {
  it('Saber opens 142\'s Approvals, then 149\'s, then 142\'s again: 142\'s "since" is his last 142 visit', () => {
    signInAs('ad', 'a')
    lwInitStore(memoryBackend(), { unit: 'u142' }); expect(worldUnit()).toBe('u142')
    openApprovals('stiff', 1_000)
    expect(apprSeenAt('stiff'), 'setup: the visit was stored').toBe(1_000)
    /* step 5's pick (a master admin, D2197) — or, through pageUnit('own'), an admin with Plan of 149 (D2201, D2241) */
    lwInitStore(memoryBackend(), { unit: 'u149' }); expect(worldUnit()).toBe('u149')
    openApprovals('stiff', 2_000)
    lwInitStore(memoryBackend(), { unit: 'u142' })
    expect(openApprovals('stiff', 3_000), "142's previous visit, not 149's").toBe(1_000)
  })
})

describe('O3 (GUARD — green today; keep it green through PE1) — the member view is a member\'s (D292)', () => {
  it('Saber, then Hunter holding Plan of 149, switched to the member view: no edit mode', () => {

describe('O4 — the one OIL list saves the answers of the squadron it LISTS (PB4 must reach ui/oilday.ts saveOilDay)', () => {
  it('a 149 input on Sat 18 Jul, listed for 149, is answered by the master admin whose own squadron is 142', () => {
    lwInitStore(memoryBackend()); setPeople(projectPeople()); wireLeaveWarSync()
    signInAs('ad', 'a')
    const SAT = '2026-07-18'
    const row = { person: 'osp9', date: 'Jul 18', yr: 2026, allday: true, type: 'Duty', title: 'O4 weekend duty 149', remarks: '', by: 'stiff', at: Date.now(), iid: 'o4w149', ord: 99001 }
    expect(writeInputsBatch(() => { INPUTS.push(row as any) }), 'setup: filed by the master admin for Osprey (149)').toBe(true)
    expect(oilOpenOn(SAT, SAT, 'u149').map(o => o.iid), "setup: 149's list holds it").toContain('o4w149')
    expect(oilOpenOn(SAT, SAT, 'u142').map(o => o.iid), "setup: 142's list does not").not.toContain('o4w149')
    /* the list opened on 149 (PB4: a 149 holiday line tapped by an admin of 149 whose own page is 142's); its save is
       handed the squadron it listed — today's saveOilDay ignores it and filters by myUnit(), the reader's home */
    const n = (saveOilDay as any)(SAT, SAT, { o4w149: 'y' }, 'u149')
    const rec: any = INPUTS.find((x: any) => x.iid === 'o4w149')
    expect({ n, oil: rec && rec.oil }, 'one answered, its Saturday credited').toEqual({ n: 1, oil: { [SAT]: 1 } })
  })
})
