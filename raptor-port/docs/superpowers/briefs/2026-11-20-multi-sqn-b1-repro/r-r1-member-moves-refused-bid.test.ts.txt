const dayRecs = (d: string): any[] => {
  const out: any[] = []
  const walk = (o: any) => {
    if (!o || typeof o !== 'object') return
    for (const [k, v] of Object.entries(o)) {
      if (k === d && Array.isArray(v)) out.push(...(v as any[]))
      else if (typeof v === 'object') walk(v)
    }
  }
  walk((getState() as any).wars)
  return out.filter(r => r && r.kind === 'request')
}
/* Saber opens 20 Jul–10 Aug; Ranger bids LL on 28 Jul; Saber refuses it (refuse=false: left pending) */
const bidOn28 = (refuse: boolean) => {
  signInAs('ad', 'a')
  openDates('2026-07-20', '2026-08-10')
  signInAs('us', 'us')
  expect(setCell('bane', '2026-07-28', 'LL')).toBe(true)
  if (refuse) { signInAs('ad', 'a'); setBidStates([{ personId: 'bane', date: '2026-07-28' }], 'refused'); signInAs('us', 'us') }
  const rec = dayRecs('2026-07-28')[0]
  expect(rec?.state).toBe(refuse ? 'refused' : 'pending')
  return rec
}

describe('R1 — a member and his own refused bid (D1177: a member moves his own bid while it waits)', () => {
  it('no Move is offered to the member on his refused bid', () => {
    bidOn28(true)
    expect(movableRecords([{ personId: 'bane', date: '2026-07-28' }]), 'Move offered on a refused bid').toEqual([])
  })
  it('forced, the move answers a refusal — never "moved" — and changes nothing', () => {
    const rec = bidOn28(true)
    const res = moveRecords([{ personId: 'bane', date: '2026-07-28', id: rec.id, kind: 'request' }], 1)
    expect(dayRecs('2026-07-28')[0]?.state, 'the refused bid stays').toBe('refused')
    expect(dayRecs('2026-07-29'), 'nothing landed').toEqual([])
    expect(res, 'said "moved" while nothing moved').not.toBe('moved')
  })
  it('(control) his pending bid is offered and moves, landing pending on the 29th', () => {
    const rec = bidOn28(false)
    expect(movableRecords([{ personId: 'bane', date: '2026-07-28' }]).map(m => m.id)).toEqual([rec.id])
    expect(moveRecords([{ personId: 'bane', date: '2026-07-28', id: rec.id, kind: 'request' }], 1)).toBe('moved')
    expect(dayRecs('2026-07-29')[0]?.state).toBe('pending')
  })
  it('(control) the admin moves the refused bid: it lands pending on the 29th', () => {
    const rec = bidOn28(true)
    signInAs('ad', 'a')
    expect(moveRecords([{ personId: 'bane', date: '2026-07-28', id: rec.id, kind: 'request' }], 1)).toBe('moved')
    expect(dayRecs('2026-07-29')[0]?.state).toBe('pending')
  })
})
