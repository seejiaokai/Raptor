// @vitest-environment jsdom
/* B7 (Opus B2): "Request access" names no squadron, so every squadron's admins are told of every request and the
   approver's squadron takes him. D2094: "'Request access' asks the person's squadron and the request goes to that
   squadron's admins only"; D2111: it asks which squadron, the first choice on its card. Batch 1 §3.1 lists it; its
   step 5 builds it. The proof: a request made for 142 never reaches 149's admin, and is not approved into 149. */
import { describe, expect, it } from 'vitest'
import { useWorld, signInAs, signOut } from './_harness'
import { requestAccess, unseenRequests, ACCESS_REQS, approveRequestNew } from '../state/accounts'
import { homeOn } from '../state/units'
import { PEOPLE } from '../engine/people'

useWorld()

describe("B7 — a request goes to its own squadron's admins only", () => {
  it('the stored request names a squadron', () => {
    signInAs('newbie@x', 'p')
    expect(requestAccess({ cs: 'Newbie', ini: 'NB', seat: 'FCP', cat: 'C', unit: 'u142' } as any)).toBeNull()
    expect((ACCESS_REQS[0] as any).unit, 'the request carries the squadron he asked').toBe('u142')
  })
  it("a request for 142 (or, as built, with no squadron at all) is not shown to 149's admin", () => {
    signInAs('newbie@x', 'p')
    expect(requestAccess({ cs: 'Newbie', ini: 'NB', seat: 'FCP', cat: 'C', unit: 'u142' } as any)).toBeNull()
    signOut()
    signInAs('ad2', 'a2')
    expect(unseenRequests().length, "142's admin is told").toBe(1)
    signOut()
    signInAs('ad9', 'a9')
    expect(unseenRequests().length, "149's admin is not").toBe(0)
  })
  it("149's admin cannot approve a 142 request into 149", () => {
    signInAs('newbie@x', 'p')
    expect(requestAccess({ cs: 'Newbie', ini: 'NB', seat: 'FCP', cat: 'C', unit: 'u142' } as any)).toBeNull()
    signOut()
    signInAs('ad9', 'a9')
    const rq = ACCESS_REQS[0]!
    approveRequestNew(rq.id, { cs: 'Newbie', ini: 'NB', seat: 'FCP', cat: 'C' } as any, 'main')
    const pid = Object.keys(PEOPLE).find(id => (PEOPLE as any)[id].cs === 'Newbie')
    expect(pid ? homeOn(pid) : 'not added', 'never 149').not.toBe('u149')
  })
})
