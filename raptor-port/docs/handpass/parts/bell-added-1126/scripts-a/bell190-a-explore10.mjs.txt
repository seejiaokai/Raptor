// bell190 walker A — exploration 10: what a press on the input's own label in Personal Inputs opens (week)
import * as L from './bell190-a-lib.mjs'
console.log('bell190-a-explore10 v1')
const w = await L.newWorld('desk')
await L.signIn(w, 'ad')
const p = w.page
const iid = await L.fileInput(w, L.WED, { type: 'Training', people: [L.ID.Tally], from: '09:00', to: '11:00' })
await L.go(p, 'editsched')
const fold = p.locator('#eWeek [data-pitog="2"]').filter({ visible: true }).first()
console.log('fold', await fold.count())
await fold.scrollIntoViewIfNeeded(); await fold.click(); await L.sleep(600)
const labs = await p.evaluate(iid => [...document.querySelectorAll(`[data-inpedit="${iid}"]`)].map(e => { const r = e.getBoundingClientRect(); const h = document.elementFromPoint(r.left + r.width / 2, r.top + r.height / 2); return `${e.tagName}.${e.className} in=${e.closest('[id]')?.id} ${Math.round(r.left)},${Math.round(r.top)} ${Math.round(r.width)}x${Math.round(r.height)} hit=${h === e || e.contains(h)} t="${e.textContent.trim()}"` }), iid)
console.log(L.J(labs, null, 1))
await L.shot(w, 'explore10-unfolded')
const lab = p.locator(`#eWeek [data-inpedit="${iid}"]`).filter({ visible: true }).first()
await lab.scrollIntoViewIfNeeded(); await L.sleep(300); await lab.click(); await L.sleep(1000)
console.log('after click windows', L.J(await p.evaluate(() => [...document.querySelectorAll('[data-testid^="win-"], #inpEditPop, [role="dialog"]')].filter(e => e.getBoundingClientRect().width > 3).map(e => (e.getAttribute('data-testid') || e.id) + ' ' + (e.innerText || '').replace(/\s+/g, ' ').slice(0, 120)))))
console.log('toast', await L.toast(p))
console.log('person options', L.J(await p.evaluate(() => [...document.querySelector('#inpEditPop #inpEditPerson').options].map(o => o.value + '=' + o.textContent).slice(-6))), 'text', (await p.evaluate(() => document.querySelector('#inpEditPop').innerText)).replace(/\n/g, ' | ').slice(0, 300))
console.log('window controls', L.J(await p.evaluate(() => [...document.querySelectorAll('#inpEditPop button, #inpEditPop select, #inpEditPop input')].filter(e => e.getBoundingClientRect().width > 0).map(e => `${e.tagName} ${e.id || ''} ${[...e.attributes].filter(a => a.name.startsWith('data-') || a.name === 'aria-pressed').map(a => a.name + '=' + a.value).join(' ')} "${(e.textContent || e.value || '').trim().slice(0, 20)}"`).slice(0, 90))))
await L.shot(w, 'explore10-clicked')
await L.closeAll()
