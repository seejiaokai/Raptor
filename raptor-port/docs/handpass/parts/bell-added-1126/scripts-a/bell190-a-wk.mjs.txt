// bell190 walker A — does WebKit open the served app at all (timings)
import { createRequire } from 'module'
const require = createRequire(process.cwd() + '/package.json')
const { webkit } = require('playwright')
console.log('bell190-a-wk v1')
const t0 = Date.now(); const T = s => console.log(((Date.now() - t0) / 1000).toFixed(1), s)
const b = await webkit.launch(); T('launched')
const ctx = await b.newContext({ viewport: { width: 402, height: 874 }, deviceScaleFactor: 3, isMobile: true, hasTouch: true }); T('context')
const p = await ctx.newPage(); T('page')
p.on('console', m => T('console ' + m.type() + ' ' + m.text().slice(0, 120)))
p.on('pageerror', e => T('pageerror ' + String(e).slice(0, 160)))
p.on('requestfailed', r => T('failed ' + r.url()))
await p.goto('http://localhost:4190/', { waitUntil: 'commit', timeout: 30000 }).then(() => T('commit')).catch(e => T('goto err ' + e.message.split('\n')[0]))
await p.waitForSelector('#luser', { timeout: 40000 }).then(() => T('luser')).catch(e => T('no luser ' + e.message.split('\n')[0]))
await b.close(); T('closed')
