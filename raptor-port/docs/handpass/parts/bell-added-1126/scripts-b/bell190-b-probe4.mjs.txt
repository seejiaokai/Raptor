// bell190-b probe 4: add a third man on Admin -> Users; file one Training for Ranger from the List; read NOTICES and Ranger's bell.
import { createRequire } from 'module'
const require = createRequire(process.cwd() + '/package.json')
const { chromium } = require('playwright')
console.log('bell190-b-probe4 v1')
const URL = 'http://localhost:4190/'
const S = process.argv[2]
const b = await chromium.launch()
const ctx = await b.newContext({ viewport: { width: 1536, height: 864 } })
const page = await ctx.newPage()
const errs = []; page.on('pageerror', e => errs.push(String(e)))
async function signIn(u, p) {
  await page.goto(URL); await page.waitForSelector('#luser')
  await page.fill('#luser', u); await page.fill('#lpass', p)
  await page.click('#loginForm button[type=submit]'); await page.waitForSelector('#notifyBell'); await page.waitForTimeout(500)
}
await signIn('ad', 'a')
await page.evaluate(() => window.go('admin')); await page.waitForTimeout(800)
console.log('page', await page.evaluate(() => window.CURPAGE))
const usersTab = await page.evaluate(() => [...document.querySelectorAll('button')].filter(e => e.getBoundingClientRect().width > 0).map(e => e.id + ':' + e.innerText.trim().slice(0, 20)).join(' | '))
console.log(usersTab.slice(0, 1500))
const hasAdd = await page.locator('#accAddCs').count()
console.log('accAddCs present', hasAdd)
if (hasAdd) {
  console.log('seat options', await page.evaluate(() => [...document.querySelector('#accAddSeat').options].map(o => o.value + ':' + o.text).join(',')))
  await page.fill('#accAddCs', 'Kestrel'); await page.fill('#accAddIni', 'KS')
  await page.selectOption('#accAddSeat', { label: 'WSO' }).catch(async () => console.log('no WSO label'))
  console.log('cat options', await page.evaluate(() => [...document.querySelector('#accAddCat').options].map(o => o.value).join(',')))
  const cat = await page.evaluate(() => [...document.querySelector('#accAddCat').options].map(o => o.value).filter(Boolean)[0])
  await page.selectOption('#accAddCat', cat)
  await page.fill('#accAddName', 'kestrel@mail')
  await page.waitForTimeout(200)
  console.log('role options', await page.evaluate(() => { const s = document.querySelector('#accAddRole'); return s ? [...s.options].map(o => o.value).join(',') : 'none' }))
  await page.click('#accAdd'); await page.waitForTimeout(600)
  console.log('added?', await page.evaluate(() => Object.keys(window.PEOPLE).filter(k => window.PEOPLE[k].cs === 'Kestrel')))
  await page.screenshot({ path: S + '/probe4-admin.png' })
}
await page.evaluate(() => window.go('inputs')); await page.waitForTimeout(700)
await page.click('#inListBtn'); await page.waitForTimeout(400)
await page.click('#inNew'); await page.waitForTimeout(500)
await page.selectOption('#inpEditType', 'Training')
await page.waitForTimeout(200)
await page.selectOption('#inpEditPerson', 'bane')
await page.click('[data-testid="inpEdCal-day-2026-07-15"]'); await page.waitForTimeout(300)
await page.screenshot({ path: S + '/probe4-win-after-15.png' })
const win = await page.evaluate(() => document.querySelector('[data-testid="win-inputedit"]').innerText.replace(/\s+/g, ' ').slice(0, 600))
console.log('window text', win)
const had = await page.evaluate(() => window.INPUTS.map(r => r.iid))
await page.click('#inpEditSave'); await page.waitForTimeout(700)
console.log('still open?', await page.locator('[data-testid="win-inputedit"]').count())
console.log('new', await page.evaluate(had => JSON.stringify(window.INPUTS.filter(r => !had.includes(r.iid))), had))
console.log('NOTICES', await page.evaluate(() => JSON.stringify(window.NOTICES).slice(0, 2000)))
await page.click('#logout'); await page.waitForSelector('#luser')
await signIn('us', 'us')
await page.screenshot({ path: S + '/probe4-ranger-top.png' })
console.log('bell btn', await page.evaluate(() => document.querySelector('#notifyBell').outerHTML.slice(0, 500)))
await page.click('#notifyBell'); await page.waitForTimeout(500)
console.log('bell list html', await page.evaluate(() => { const l = document.querySelector('.bell-list'); return l ? l.outerHTML.slice(0, 2500) : 'NO LIST ' + document.body.innerText.slice(0, 200) }))
await page.screenshot({ path: S + '/probe4-ranger-bell.png' })
const first = page.locator('.bell-list > *').first()
if (await first.count()) { await first.click(); await page.waitForTimeout(800); await page.screenshot({ path: S + '/probe4-ranger-tap.png' }); console.log('after tap page', await page.evaluate(() => window.CURPAGE), await page.locator('[data-testid="win-inputedit"]').count()) }
// kestrel sign-in
await page.click('#logout').catch(() => {}); await page.waitForSelector('#luser')
await signIn('kestrel@mail', 'x').catch(e => console.log('kestrel sign-in failed', String(e).slice(0, 200)))
console.log('kestrel signed in as', await page.evaluate(() => document.querySelector('#roleBadge')?.innerText))
await b.close()
console.log('errors', errs)
