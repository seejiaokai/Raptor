// bell190-c S6 (desktop): Undo of leave entered with Auto-approve ON — (a) before Ranger reads it, then Redo; (b) after he
// read it. Undo's list clears at sign-out (D148), so (b) needs Saber still signed in: a second tab of the same browser is
// Ranger's — the walk-scripts §3 two-tab caveat applies and is reported. [F45, F46, A38]
const L = await import('./bell190-c-lib.mjs')
console.log('bell190-c-s6 v1')
const w = await L.newWorld('desk')
const P = w.page
const lineOf = async (pg, day) => (await pg.evaluate(() => JSON.parse(JSON.stringify(window.NOTICES || [])))).filter(n => n.to === 'bane' && n.day === day).map(L.fmt)
await L.signIn(w, 'ad', 'a'); await L.toWar(w)
// (a)
await L.openCell(w, 'bane', '2026-07-22'); console.log('auto', await L.setAuto(w, true)); console.log('place', JSON.stringify(await L.place(w, 'LL')))
console.log('(a) after place: cell', await L.cellText(w, 'bane', '2026-07-22'), 'lines', JSON.stringify(await lineOf(P, '2026-07-22')))
await L.undo(w)
console.log('(a) after Undo: cell', JSON.stringify(await L.cellText(w, 'bane', '2026-07-22')), 'lines', JSON.stringify(await lineOf(P, '2026-07-22')))
await L.shot(w, 'c-s6-01-after-undo')
await L.tap(w, '#redoBtn'); await P.waitForTimeout(700)
console.log('(a) after Redo: cell', JSON.stringify(await L.cellText(w, 'bane', '2026-07-22')), 'lines', JSON.stringify(await lineOf(P, '2026-07-22')))
await L.undo(w)
console.log('(a) after a second Undo: cell', JSON.stringify(await L.cellText(w, 'bane', '2026-07-22')), 'lines', JSON.stringify(await lineOf(P, '2026-07-22')))
// (b) LL on 23 Jul, Ranger reads it in a second tab, Saber's tab presses Undo
await L.openCell(w, 'bane', '2026-07-23'); await L.setAuto(w, true); console.log('place 23', JSON.stringify(await L.place(w, 'LL')))
console.log('(b) after place: lines', JSON.stringify(await lineOf(P, '2026-07-23')))
const p2 = await w.ctx.newPage()
const w2 = { ctx: w.ctx, page: p2, errs: [], phone: false }
await p2.goto(L.URL, { waitUntil: 'domcontentloaded' }); await p2.waitForTimeout(1500)
const tab2 = await p2.evaluate(() => ({ card: !!document.querySelector('#luser') && document.querySelector('#luser').getBoundingClientRect().height > 0, bell: !!document.querySelector('#notifyBell') }))
console.log('tab 2 opens on:', JSON.stringify(tab2))
if (tab2.card) await L.signIn(w2, 'us', 'us')
else console.log('tab 2 is already signed in as:', await p2.evaluate(() => document.querySelector('.who, #whoami, .abtn.who')?.innerText || document.body.innerText.match(/[A-Z]+ · (ADMIN|MEMBER)/)?.[0]))
const r1 = await L.bell(w2, 'c-s6-02-ranger-reads')
console.log('(b) Ranger reads (tab 2):', JSON.stringify(r1.lines.map(l => [l.unread, l.text])))
await L.tap(w2, '#notifyBell'); await p2.waitForTimeout(400)
const r1b = await L.bell(w2, null); await L.tap(w2, '#notifyBell')
console.log('(b) Ranger bell reopened (read marks):', JSON.stringify(r1b.lines.map(l => [l.unread, l.text])))
// Saber's tab: is it still Saber? its Undo still offered?
await P.bringToFront()
console.log('tab 1 still shows:', await P.evaluate(() => document.body.innerText.match(/[A-Z]+ · (ADMIN|MEMBER)/)?.[0]), 'undo enabled:', await P.evaluate(() => !document.querySelector('#undoBtn').disabled))
await L.undo(w)
console.log('(b) after Undo in tab 1: cell', JSON.stringify(await L.cellText(w, 'bane', '2026-07-23')), 'tab-1 store lines', JSON.stringify(await lineOf(P, '2026-07-23')))
await L.shot(w, 'c-s6-03-tab1-after-undo')
// Ranger's tab reloaded (reads the stored records afresh)
await p2.reload({ waitUntil: 'domcontentloaded' }); await p2.waitForTimeout(1500)
if (await L.vis(p2, '#luser')) await L.signIn(w2, 'us', 'us')
const r2 = await L.bell(w2, 'c-s6-04-ranger-after-undo')
console.log('(b) Ranger bell after the Undo (tab 2 reloaded):', JSON.stringify({ lit: r2.lit && r2.lit.cls, lines: r2.lines.map(l => [l.unread, l.text]), list: r2.list }))
console.log('(b) tab-2 store lines for 23 Jul:', JSON.stringify(await lineOf(p2, '2026-07-23')))
await L.tap(w2, '#notifyBell').catch(() => {})
// and in one tab: tab 2 closed, tab 1 signs out, Ranger signs in there
await p2.close()
await L.signOut(w); await L.signIn(w, 'us', 'us')
const r3 = await L.bell(w, 'c-s6-05-ranger-one-tab')
console.log('(b) Ranger bell, one tab, fresh sign-in:', JSON.stringify({ lit: r3.lit && r3.lit.cls, lines: r3.lines.map(l => [l.unread, l.text]), list: r3.list }))
for (const l of r3.lines) if (/23 Jul/.test(l.text)) { console.log('tap 23 Jul line ->', JSON.stringify(await L.tapLine(w, l.id, 'c-s6-06-tap-23jul-line'))); break }
console.log('errors', w.errs)
await L.closeAll()
