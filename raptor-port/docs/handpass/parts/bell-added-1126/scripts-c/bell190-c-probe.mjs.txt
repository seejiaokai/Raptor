// bell190-c probe: the bundle, the people, the Leave War rows, the bid window's controls on Ranger's day.
import { createRequire } from 'module'
const require = createRequire(process.cwd() + '/package.json')
const { chromium } = require('playwright')
console.log('bell190-c-probe v1')
const URL = 'http://localhost:4190/'
const OUT = process.argv[2]
const b = await chromium.launch()
const ctx = await b.newContext({ viewport: { width: 1536, height: 864 } })
const page = await ctx.newPage()
const errs = []; page.on('pageerror', e => errs.push(String(e))); page.on('console', m => { if (m.type() === 'error') errs.push(m.text()) })
const scripts = []; page.on('response', r => { if (/\/assets\/index-.*\.js$/.test(r.url())) scripts.push(r.url()) })
await page.goto(URL); await page.waitForSelector('#luser')
await page.fill('#luser', 'ad'); await page.fill('#lpass', 'a')
await page.click('#loginForm button[type=submit]'); await page.waitForSelector('#notifyBell')
console.log('bundle', scripts)
const ppl = await page.evaluate(() => Object.entries(window.PEOPLE || {}).filter(([k, v]) => v && !v.deleted).map(([k, v]) => k + '=' + v.cs + (v.role ? '/' + v.role : '')).join(' '))
console.log('people', ppl)
const acc = await page.evaluate(() => JSON.stringify(window.ACCOUNTS || window.USERS || null).slice(0, 600))
console.log('accounts', acc)
await page.evaluate(() => window.go('leavewar'))
await page.waitForTimeout(3000)
const rows = await page.evaluate(() => [...document.querySelectorAll('[data-testid^="row-"]')].map(e => e.dataset.testid + ':' + (e.innerText || '').replace(/\s+/g, ' ').trim().slice(0, 30)).join(' | '))
console.log('rows', rows.slice(0, 2500))
const months = await page.evaluate(() => [...document.querySelectorAll('[data-testid^="month-"]')].map(e => e.dataset.testid).join(' '))
console.log('months', months)
const tabs = await page.evaluate(() => [...document.querySelectorAll('[data-testid^="lw-tab-"]')].map(e => e.dataset.testid + ':' + e.innerText.trim()).join(' | '))
console.log('tabs', tabs)
const some = await page.evaluate(() => [...document.querySelectorAll('td[data-testid^="cell-"]')].slice(0, 3).map(e => e.dataset.testid).join(' '))
console.log('cells', some)
await page.screenshot({ path: OUT + '/probe-war.png' })
console.log('notices', await page.evaluate(() => JSON.stringify((window.NOTICES || []).slice(-3))))
console.log('errors', errs)
await b.close()
